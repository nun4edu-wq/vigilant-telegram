                'سابع': 7, 'السابع': 7,
                'ثامن': 8, 'الثامن': 8,
                'تاسع': 9, 'التاسع': 9,
                'عاشر': 10, 'العاشر': 10
            };

            classesArray.sort((a, b) => {
                let nameA = a.name.trim();
                let nameB = b.name.trim();

                let firstWordA = nameA.split(/[\s\/\-\(\)]+/)[0];
                let firstWordB = nameB.split(/[\s\/\-\(\)]+/)[0];

                let rankA = rankOrder[firstWordA] || 999;
                let rankB = rankOrder[firstWordB] || 999;

                if (rankA !== rankB) {
                    return rankA - rankB;
                }
                return nameA.localeCompare(nameB, 'ar', { numeric: true });
            });
        }

        function onTermChange() {
            const termVal = document.getElementById('termSelect').value;
            const weekSelect = document.getElementById('weekSelect');
            if(!weekSelect) return;
            weekSelect.innerHTML = '';
            
            let allOpt = document.createElement('option');
            allOpt.value = 'ALL_WEEKS';
            allOpt.innerText = '📅 جميع الأسابيع (تراكمي)';
            weekSelect.appendChild(allOpt);

            let weeksList = [];
            if(termVal === 'ALL') {
                weeksList = [...schoolWeeksData.term1, ...schoolWeeksData.term2];
            } else {
                weeksList = termVal === '1' ? schoolWeeksData.term1 : schoolWeeksData.term2;
            }

            let currentWeekParsed = parseWeekCode(currentActiveWeekCode);

            weeksList.forEach((w, idx) => {
                let termNum = termVal === 'ALL' ? (idx < schoolWeeksData.term1.length ? 1 : 2) : parseInt(termVal);
                let wCode = `t${termNum}_w${w.week}_${idx}`;

                if (currentUser && (currentUser.role === 'SCHOOL' || currentUser.role === 'MANAGER')) {
                    if (isFutureWeek(termNum, w.week, currentWeekParsed)) {
                        return;
                    }
                }

                let opt = document.createElement('option');
                opt.value = wCode;
                opt.innerText = w.label;
                weekSelect.appendChild(opt);
            });

            weekSelect.value = currentActiveWeekCode;
            if(!weekSelect.value) weekSelect.selectedIndex = 0;
            onWeekChange();
        }

        function parseWeekCode(code) {
            let parts = code.split('_');
            return { term: parseInt(parts[0].replace('t','')), week: parseInt(parts[1].replace('w','')) };
        }

        function isFutureWeek(term, week, current) {
            if (term > current.term) return true;
            if (term === current.term && week > current.week) return true;
            return false;
        }

        function onWeekChange() {
            const weekSelect = document.getElementById('weekSelect');
            if(!weekSelect || !weekSelect.value) return;
            let weekVal = weekSelect.value;
            
            let sSun = document.getElementById('date_sun');
            let sThu = document.getElementById('date_thu');

            if(weekVal === 'ALL_WEEKS') {
                if(sSun) sSun.value = '2026-08-23';
                if(sThu) sThu.value = '2027-06-24';
                currentPage = 1;
                updateDatesHeader();
                return;
            }

            let parts = weekVal.split('_');
            let termNum = parts[0].replace('t', '');
            let weekNum = parseInt(parts[1].replace('w', ''));

            let weeksList = termNum === '1' ? schoolWeeksData.term1 : schoolWeeksData.term2;
            let wkObj = weeksList.find(w => w.week === weekNum);
            if(wkObj) {
                if(sSun) sSun.value = wkObj.start;
                if(sThu) sThu.value = wkObj.end;
                currentPage = 1;
                updateDatesHeader();
            }
        }

        function updateDatesHeader() {
            let startInput = document.getElementById('date_sun');
            let endInput = document.getElementById('date_thu');
            if(!startInput || !startInput.value) return;
            let startDate = new Date(startInput.value);
            let endDate = (endInput && endInput.value) ? new Date(endInput.value) : new Date(startDate);

            const arabicDays = ['الأحد', 'الاثنين', 'الثلاثاء', 'الأربعاء', 'الخميس', 'الجمعة', 'السبت'];
            const dayKeys = ['sun', 'mon', 'tue', 'wed', 'thu'];
            
            let startDayName = arabicDays[startDate.getDay()];
            let endDayName = arabicDays[endDate.getDay()];
            let lblStart = document.getElementById('label_date_start');
            let lblEnd = document.getElementById('label_date_end');
            if(lblStart) lblStart.innerText = `${startDayName}:`;
            if(lblEnd) lblEnd.innerText = `${endDayName}:`;

            dayKeys.forEach((key, index) => {
                let d = new Date(startDate);
                d.setDate(d.getDate() + ((currentPage - 1) * 5) + index);
                
                let dayName = arabicDays[d.getDay()]; 
                let formattedDate = `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()}`;
                
                let thTitle = document.getElementById(`th_title_${key}`);
                let dashTitle = document.getElementById(`dash_title_${key}`);
                let stHeader = document.getElementById(`st_h_${index+1}`);
                
                if(thTitle) thTitle.innerText = dayName;
                if(dashTitle) dashTitle.innerText = dayName;
                if(stHeader) stHeader.innerText = dayName;
                
                let hdrElem = document.getElementById(`hdr_date_${key}`);
                let dashElem = document.getElementById(`dash_date_${key}`);
                if(hdrElem) hdrElem.innerText = formattedDate;
                if(dashElem) dashElem.innerText = formattedDate;
            });

            renderPaginationControls(startDate, endDate);
            checkWeekPermissions();
            if(currentUser) {
                let dashView = document.getElementById('dashboardView');
                if(dashView && dashView.style.display !== 'none') {
                    renderDashboard();
                } else if(currentActiveSchoolId) {
                    calculateStats();
                }
            }
        }

        function renderPaginationControls(startDate, endDate) {
            let diffTime = Math.abs(endDate - startDate);
            let diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24)) + 1;
            let totalPages = Math.max(1, Math.ceil(diffDays / 5));

            let box = document.getElementById('paginationBox');
            if(!box) return;
            box.innerHTML = '';

            for(let i = 1; i <= totalPages; i++) {
                let btn = document.createElement('button');
                btn.className = `page-btn ${currentPage === i ? 'active' : ''}`;
                btn.innerText = i;
                btn.onclick = () => {
                    currentPage = i;
                    updateDatesHeader();
                };
                box.appendChild(btn);
            }
        }

        function checkWeekPermissions() {
            let weekSel = document.getElementById('weekSelect');
            if(!weekSel) return true;
            let selectedWeek = weekSel.value;
            let banner = document.getElementById('alert-banner');
            let actionBox = document.getElementById('school-action-box');
            let actionTh = document.getElementById('th-action-col');
            let saveGlobalBtn = document.getElementById('btn-save-global');

            let allowEdit = true;

            if (currentUser && currentUser.role === 'MANAGER') {
                allowEdit = false;
                if(banner) {
                    banner.style.display = 'block';
                    banner.innerText = '👁️ صلاحية مدير مجمع: استعراض بيانات مدارس مجمعك فقط (للقراءة فقط).';
                }
            } else if (currentUser && currentUser.role === 'SUPER_ADMIN') {
                allowEdit = true;
                if(banner) banner.style.display = 'none';
            } else if (selectedWeek === 'ALL_WEEKS') {
                allowEdit = false;
                if(banner) {
                    banner.style.display = 'block';
                    banner.innerText = '📊 عرض تراكمي لجميع الأسابيع (وضع القراءة فقط).';
                }
            } else if (selectedWeek !== currentActiveWeekCode) {
                if (currentUser && currentUser.allowPastEdit) {
                    allowEdit = true;
                    if(banner) {
                        banner.style.display = 'block';
                        banner.innerText = '🔓 ملاحظة: تمتلك صلاحية استثنائية بالتعديل في الأسابيع السابقة.';
                    }
                } else {
                    allowEdit = false;
                    if(banner) {
                        banner.style.display = 'block';
                        banner.innerText = '⚠️ تنبيه: أنت تستعرض أسبوعاً سابقاً. كافة الحقول مقفلة تماماً (للقراءة فقط).';
                    }
                }
            } else {
                if(banner) banner.style.display = 'none';
                if(currentUser && currentUser.role !== 'SCHOOL') {
                    allowEdit = false; 
                }
            }

            let isSuperAdmin = currentUser && currentUser.role === 'SUPER_ADMIN';
            if(actionBox) actionBox.style.display = (allowEdit && isSuperAdmin) ? 'flex' : 'none';
            if(actionTh) actionTh.style.display = (allowEdit && isSuperAdmin) ? 'table-cell' : 'none';
            if(saveGlobalBtn) saveGlobalBtn.style.display = allowEdit ? 'inline-block' : 'none';

            return allowEdit;
        }

        function handleLogin(e) {
            e.preventDefault();
            const uInput = document.getElementById('login-user');
            const pInput = document.getElementById('login-pass');
            if(!uInput || !pInput) return;
            const u = uInput.value.trim().toLowerCase();
            const p = pInput.value.trim();
            const found = users.find(x => x.username.toLowerCase() === u && x.pass === p);
            if(found) {
                currentUser = found;
                let overlay = document.getElementById('login-overlay');
                let sidebar = document.getElementById('sidebar-nav');
                let topbar = document.getElementById('topbar-header');
                let contentArea = document.getElementById('content-area-main');
                let badge = document.getElementById('user-badge-info');

                if(overlay) overlay.style.display = 'none';
                if(sidebar) sidebar.style.display = 'flex';
                if(topbar) topbar.style.display = 'flex';
                if(contentArea) contentArea.style.display = 'block';
                if(badge) badge.innerText = `${currentUser.name} (${getRoleTitle(currentUser.role)})`;
                
                applyRoleFiltersOnLogin();
                setupView();
            } else {
                alert('⚠️ اسم المستخدم أو كلمة المرور غير صحيحة.');
            }
        }

        function applyRoleFiltersOnLogin() {
            let branchSel = document.getElementById('sb-branch');
            let sectionSel = document.getElementById('sb-section');
            let gradeSel = document.getElementById('sb-grade');

            if(branchSel) { branchSel.value = 'ALL'; branchSel.disabled = false; }
            if(sectionSel) { sectionSel.value = 'ALL'; sectionSel.disabled = false; }
            if(gradeSel) { gradeSel.value = 'ALL'; gradeSel.disabled = false; }
            onSectionFilterChange();

            if(currentUser.role === 'SCHOOL' && currentUser.schoolId) {
                let school = schoolsData.find(s => s.id === currentUser.schoolId);
                if(school) {
                    if(branchSel) { branchSel.value = school.branch; branchSel.disabled = true; }
                    if(sectionSel) { sectionSel.value = school.section; sectionSel.disabled = true; onSectionFilterChange(); }
                    if(gradeSel) { gradeSel.value = school.grade; gradeSel.disabled = true; }
                }
            } else if(currentUser.role === 'MANAGER' && currentUser.branch) {
                if(branchSel) { branchSel.value = currentUser.branch; branchSel.disabled = true; }
            } else if(currentUser.role === 'SUPERVISOR' && currentUser.section) {
                if(sectionSel) { sectionSel.value = currentUser.section; sectionSel.disabled = true; }
            }
            onTermChange();
        }

            currentActiveSchoolId = schoolId;

            if(dashView) dashView.style.display = 'none';
            if(schoolView) schoolView.style.display = 'block';

                let branchSel = document.getElementById('sb-branch');

            if(!school.absent) school.absent = {sun:0, mon:0, tue:0, wed:0, thu:0};

            sortClassesProperly(school.classes);
            updateDayButtonsUI(school);

            let isSuperAdmin = currentUser && currentUser.role === 'SUPER_ADMIN';

            school.classes.forEach((cls, idx) => {
                let tr = document.createElement('tr');
                let dayKeys = ['sun', 'mon', 'tue', 'wed', 'thu'];
                
                let tdsHtml = '';
                dayKeys.forEach(dk => {
                    let isHolidayOrRemote = school.dayStatuses[dk] === 'holiday';
                    let isDisabledInput = isHolidayOrRemote || !canEdit;
                    let absVal = (school.classAbsent && school.classAbsent[idx] && school.classAbsent[idx][dk] !== undefined) ? school.classAbsent[idx][dk] : (idx === 0 ? (school.absent[dk] || 0) : 0);
                    
                    tdsHtml += `<td><input type="number" class="input-absent ${isDisabledInput?'input-disabled':''}" ${isDisabledInput?'disabled':''} value="${absVal}" min="0" max="${cls.students}" onfocus="handleAbsentFocus(this)" onblur="handleAbsentBlur(this)" oninput="updateClassAbsent('${schoolId}', ${idx}, '${dk}', this)"></td>`;

                tbody.appendChild(tr);
            });

        function toggleDayStatus(day) {
            let canEdit = checkWeekPermissions();
            if(!canEdit) { alert('⚠️ وضع القراءة فقط مفعل.'); return; }
            let school = schoolsData.find(s => s.id === currentActiveSchoolId);
            if(!school) return;
            if(!school.dayStatuses) school.dayStatuses = {sun:'onsite', mon:'onsite', tue:'onsite', wed:'onsite', thu:'onsite'};

            let current = school.dayStatuses[day] || 'onsite';
            if(current === 'onsite') school.dayStatuses[day] = 'remote';
            else if(current === 'remote') school.dayStatuses[day] = 'holiday';
            else school.dayStatuses[day] = 'onsite';

            saveDataToStorage();
            renderSchoolTable(currentActiveSchoolId);
        }

        function handleAbsentFocus(input) {
            if (input.value === '0') {
                input.value = '';
            }
        }

        function handleAbsentBlur(input) {
            if (input.value.trim() === '') {
                input.value = '0';
                let event = new Event('input', { bubbles: true });
                input.dispatchEvent(event);
            }
        }

        function updateClassAbsent(schoolId, classIdx, day, input) {
            let canEdit = checkWeekPermissions();
            if(!canEdit) return; 
            let school = schoolsData.find(s => s.id === schoolId);
            if(school) {
                if(!school.classAbsent) school.classAbsent = {};
                if(!school.classAbsent[classIdx]) school.classAbsent[classIdx] = {sun:0, mon:0, tue:0, wed:0, thu:0};
                
                let maxStudents = school.classes[classIdx].students;
                let val = parseInt(input.value) || 0;
                
                if(val > maxStudents) {
                    alert(`⚠️ عدد الغياب (${val}) لا يمكن أن يتجاوز عدد طلاب الفصل (${maxStudents})!`);
                    val = maxStudents;
                    input.value = maxStudents;
                }

                school.classAbsent[classIdx][day] = val;
                
                let totalDayAbsent = 0;
                school.classes.forEach((c, cIdx) => {
                    let rowAbs = (school.classAbsent[cIdx] && school.classAbsent[cIdx][day] !== undefined) ? school.classAbsent[cIdx][day] : (cIdx === 0 ? (school.absent[day] || 0) : 0);
                    totalDayAbsent += rowAbs;
                });
                school.absent[day] = totalDayAbsent;

                saveDataToStorage();
                calculateStats();
            }
        }

        function addNewClassToSchool() {
            if(!currentUser || currentUser.role !== 'SUPER_ADMIN') { alert('⚠️ عذراً، إضافة الفصول مقتصرة على المسؤول العام فقط.'); return; }
            let nameInput = document.getElementById('newClassNameInput');
            let studInput = document.getElementById('newClassStudentsInput');
            if(!nameInput) return;
            let cName = nameInput.value.trim();
            let cStudents = studInput ? (parseInt(studInput.value) || 25) : 25;
            if(!cName) { alert('يرجى إدخال اسم الفصل.'); return; }
            let targetId = currentActiveSchoolId;
            let school = schoolsData.find(s => s.id === targetId);
            if(!school) { alert('⚠️ يرجى اختيار المدرسة أولاً.'); return; }
            if(!school.classes) school.classes = [];
            
            school.classes.push({ name: cName, students: cStudents });
            sortClassesProperly(school.classes);

            saveDataToStorage();
            nameInput.value = '';
            renderSchoolTable(school.id);
        }

        function calculateStats() {
            const tbody = document.getElementById('schoolTbody');
            if(!tbody) return;
            let totalStudents = 0;
            const school = schoolsData.find(s => s.id === currentActiveSchoolId);

            Array.from(tbody.rows).forEach(row => {
                let students = parseInt(row.cells[1].children[0].value) || 0;
                totalStudents += students;
            });

            let dayKeys = ['sun', 'mon', 'tue', 'wed', 'thu'];
            let cumAbsent = 0, activeDays = 0;

            dayKeys.forEach(key => {
                let abs = school && school.absent ? (school.absent[key] || 0) : 0;
                let status = school && school.dayStatuses ? (school.dayStatuses[key] || 'onsite') : 'onsite';
                let sElem = document.getElementById(`stat_students_${key}`);
                let pElem = document.getElementById(`stat_present_${key}`);
                let aElem = document.getElementById(`stat_absent_${key}`);
                let rElem = document.getElementById(`stat_rate_${key}`);

                if(status === 'holiday') {
                    if(sElem) sElem.innerText = '-';
                    if(pElem) pElem.innerText = '-';
                    if(aElem) aElem.innerText = '-';
                    if(rElem) rElem.innerText = 'إجازة (مستبعد)';
                } else if(status === 'remote') {
                    activeDays++;
                    let pres = totalStudents;
                    let rate = '100.00 (عن بعد)';
                    if(sElem) sElem.innerText = totalStudents;
                    if(pElem) pElem.innerText = pres;
                    if(aElem) aElem.innerText = '0';
                    if(rElem) rElem.innerText = rate;
                } else {
                    activeDays++;
                    cumAbsent += abs;
                    let pres = totalStudents - abs;
                    let rate = totalStudents > 0 ? ((pres / totalStudents) * 100).toFixed(2) : '100.00';
                    if(sElem) sElem.innerText = totalStudents;
                    if(pElem) pElem.innerText = pres;
                    if(aElem) aElem.innerText = abs;
                    if(rElem) rElem.innerText = `${rate}%`;
                }
            });

            let swElem = document.getElementById('stat_students_week');
            let pwElem = document.getElementById('stat_present_week');
            let awElem = document.getElementById('stat_absent_week');
            let rwElem = document.getElementById('stat_rate_week');

            if(activeDays > 0) {
                let totalPossible = totalStudents * activeDays;
                let totalPres = totalPossible - cumAbsent;
                let weekRate = totalPossible > 0 ? ((totalPres / totalPossible) * 100).toFixed(2) : '100.00';
                if(swElem) swElem.innerText = totalStudents;
                if(pwElem) pwElem.innerText = '-';
                if(awElem) awElem.innerText = '-';
                if(rwElem) rwElem.innerText = `${weekRate}%`;
            } else {
                if(swElem) swElem.innerText = '-';
                if(pwElem) pwElem.innerText = '-';
                if(awElem) awElem.innerText = '-';
                if(rwElem) rwElem.innerText = '-';
            }
        }

        function renderDashboard() {
            let heading = document.getElementById('main-view-heading');
            if(heading) heading.innerText = 'لوحة ترتيب الأقسام والمتابعة';
            const tbody = document.getElementById('dashboardTbody');
            if(!tbody) return;
            tbody.innerHTML = '';

            let branchSel = document.getElementById('sb-branch');
            let sectionSel = document.getElementById('sb-section');
            let gradeSel = document.getElementById('sb-grade');

            let branchVal = branchSel ? branchSel.value : 'ALL';
            let sectionVal = sectionSel ? sectionSel.value : 'ALL';
            let gradeVal = gradeSel ? gradeSel.value : 'ALL';

            if (currentUser && currentUser.role === 'MANAGER' && currentUser.branch) {
                branchVal = currentUser.branch;
                if(branchSel) { branchSel.value = currentUser.branch; branchSel.disabled = true; }
            } else if (currentUser && currentUser.role === 'SUPERVISOR' && currentUser.section) {
                sectionVal = currentUser.section;
                if(sectionSel) { sectionSel.value = currentUser.section; sectionSel.disabled = true; }
            } else if (currentUser && currentUser.role === 'SCHOOL') {
                let school = schoolsData.find(s => s.id === currentUser.schoolId);
                if(school) {
                    branchVal = school.branch;
                    sectionVal = school.section;
                    gradeVal = school.grade;
                }
            }

            let list = [...schoolsData];
            if(branchVal !== 'ALL') list = list.filter(s => s.branch === branchVal);
            if(sectionVal !== 'ALL') list = list.filter(s => s.section === sectionVal);
            if(gradeVal !== 'ALL') list = list.filter(s => s.grade === gradeVal);

            let totalAllStudents = 0, totSun = 0, totMon = 0, totTue = 0, totWed = 0, totThu = 0, totAbs = 0;

            let ranked = list.map(s => {
                if(!s.classes || s.classes.length === 0) {
                    s.classes = [{name: 'أول/1', students: 25}];
                }
                if(!s.absent) s.absent = {sun:0, mon:0, tue:0, wed:0, thu:0};
                if(!s.dayStatuses) s.dayStatuses = {sun:'onsite', mon:'onsite', tue:'onsite', wed:'onsite', thu:'onsite'};

                let totalStudents = s.classes.reduce((acc, c) => acc + (parseInt(c.students) || 0), 0);
                
                let dayKeys = ['sun', 'mon', 'tue', 'wed', 'thu'];
                let totalAbsent = 0;
                let possible = 0;

                dayKeys.forEach(dk => {
                    let st = s.dayStatuses[dk] || 'onsite';
                    if(st !== 'holiday') {
                        possible += totalStudents;
                        if(st === 'onsite') {
                            totalAbsent += (s.absent[dk] || 0);
                        }
                    }
                });

                let rate = possible > 0 ? (((possible - totalAbsent) / possible) * 100).toFixed(2) : '100.00';
                return { ...s, totalStudents, absSun: s.absent.sun, absMon: s.absent.mon, absTue: s.absent.tue, absWed: s.absent.wed, absThu: s.absent.thu, totalAbsent, rate: parseFloat(rate) };
            });

            ranked.sort((a, b) => b.rate - a.rate);

            ranked.forEach((item, index) => {
                totalAllStudents += item.totalStudents;
                totSun += item.absSun; totMon += item.absMon; totTue += item.absTue; totWed += item.absWed; totThu += item.absThu; totAbs += item.totalAbsent;

                let tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><strong>${index + 1}</strong></td>
                    <td style="text-align: right; padding-right: 10px; cursor: pointer; color: #0D6EFD;" onclick="viewSchoolDetail('${item.id}')">${item.name}</td>
                    <td>${item.branch}</td>
                    <td><strong>${item.totalStudents}</strong></td>
                    <td>${item.absSun}</td>
                    <td>${item.absMon}</td>
                    <td>${item.absTue}</td>
                    <td>${item.absWed}</td>
                    <td>${item.absThu}</td>
                    <td style="color: var(--pink-accent); font-weight: bold;">${item.totalAbsent}</td>
                    <td><span style="color: var(--btn-green); font-weight: bold;">${item.rate}%</span></td>
                `;
                tbody.appendChild(tr);
            });

            let dayKeys = ['sun', 'mon', 'tue', 'wed', 'thu'];
            let dayTotals = { sun: totSun, mon: totMon, tue: totTue, wed: totWed, thu: totThu };
            dayKeys.forEach(k => {
                let abs = dayTotals[k];
                let pres = totalAllStudents - abs;
                let rate = totalAllStudents > 0 ? ((pres / totalAllStudents) * 100).toFixed(2) : '100.00';
                let dsElem = document.getElementById(`dstat_students_${k}`);
                let dpElem = document.getElementById(`dstat_present_${k}`);
                let daElem = document.getElementById(`dstat_absent_${k}`);
                let drElem = document.getElementById(`dstat_rate_${k}`);

                if(dsElem) dsElem.innerText = totalAllStudents;
                if(dpElem) dpElem.innerText = pres;
                if(daElem) daElem.innerText = abs;
                if(drElem) drElem.innerText = `${rate}%`;
            });
            let dswElem = document.getElementById('dstat_students_week');
            let dsrWeek = document.getElementById('dstat_rate_week');
            if(dswElem) dswElem.innerText = totalAllStudents;
            let grandPoss = totalAllStudents * 5;
            let grandRate = grandPoss > 0 ? (((grandPoss - totAbs) / grandPoss) * 100).toFixed(2) : '100.00';
            if(dsrWeek) dsrWeek.innerText = `${grandRate}%`;
        }

        function importSchoolClassesFromExcel(e) {
            if(!currentUser || currentUser.role !== 'SUPER_ADMIN') { alert('⚠️ عذراً، استيراد الفصول مقتصر على المسؤول العام فقط.'); return; }
            let file = e.target.files[0];
            if(!file) return;
            let reader = new FileReader();
            reader.onload = function(event) {
                try {
                    let data = new Uint8Array(event.target.result);
                    let workbook = XLSX.read(data, {type: 'array'});
                    let rows = XLSX.utils.sheet_to_json(workbook.Sheets[workbook.SheetNames[0]]);
                    let targetId = currentActiveSchoolId;
                    let school = schoolsData.find(s => s.id === targetId);
                    if(!school) { alert('⚠️ يرجى تحديد المدرسة أولاً.'); return; }
                    let newClasses = [];
                    rows.forEach(r => {
                        let cName = r["اسم الفصل"] || r["الفصل"] || r[Object.keys(r)[0]];
                        let cStud = r["عدد الطلاب"] || r["الطلاب"] || r[Object.keys(r)[1]];
                        if(cName) newClasses.push({ name: String(cName).trim(), students: parseInt(cStud) || 25 });
                    });
                    if(newClasses.length > 0) {
                        if(!school.classes) school.classes = [];
                        school.classes = newClasses;
                        sortClassesProperly(school.classes);
                        saveDataToStorage();
                        renderSchoolTable(school.id);
                        alert(`✅ تم استيراد وترتيب ${newClasses.length} فصول بنجاح.`);
                    }
                } catch(err) { alert('⚠️ خطأ في قراءة ملف الإكسل.'); }
                e.target.value = '';
            };
            reader.readAsArrayBuffer(file);
        }

        }
